1. Reach the API and dashboard

   Dashboard: {{ include "cortexops.dashboardUrl" . }}

{{- else }}
   kubectl -n {{ .Release.Namespace }} port-forward svc/{{ include "cortexops.fullname" . }}-api 8000:{{ .Values.api.service.port }}
{{- if .Values.dashboard.enabled }}
   kubectl -n {{ .Release.Namespace }} port-forward svc/{{ include "cortexops.fullname" . }}-dashboard 3000:{{ .Values.dashboard.service.port }}
   Dashboard: http://localhost:3000 (calls the API at {{ include "cortexops.apiUrl" . }})
{{- end }}

2. Create the first API key (works once per project; the raw key is shown only once)
   curl -X POST {{ include "cortexops.apiUrl" . }}/v1/keys/bootstrap \
     -H "Content-Type: application/json" \
     -d '{"project": "my-agent", "email": "you@example.com"}'

   Bootstrapped keys start on the free tier. Lift its limits (and issue further keys) with the internal key:
   INTERNAL_KEY=$(kubectl -n {{ .Release.Namespace }} get secret {{ include "cortexops.secretName" . }} -o jsonpath='{.data.INTERNAL_API_KEY}' | base64 -d)
   curl -X POST {{ include "cortexops.apiUrl" . }}/v1/admin/keys/<id from step 2>/upgrade \
     -H "X-Internal-Key: $INTERNAL_KEY" -H "Content-Type: application/json" \
     -d '{"tier": "pro", "scope": "read_write"}'

3. Point the SDK at this deployment
   export CORTEXOPS_API_URL={{ include "cortexops.apiUrl" . }}
   export CORTEXOPS_API_KEY=<raw_key from step 2>
